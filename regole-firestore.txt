rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /profili/{codice} {
      allow get: if codice.size() >= 10;
      allow create, update: if codice.size() >= 10
                            && request.resource.data.keys().hasOnly(['d','t']);
      allow list, delete: if false;
    }
  }
}
